-- admin user : 

CREATE USER 'dairy_admin'@'%' IDENTIFIED BY 'Admin@123';
GRANT ALL PRIVILEGES
ON milk_nest.*
TO 'dairy_admin'@'%';
FLUSH PRIVILEGES;

-- limited user : 

CREATE USER 'dairy_operator'@'%' IDENTIFIED BY 'Operator@123';
GRANT SELECT, INSERT, UPDATE
ON milk_nest.*
TO 'dairy_operator'@'%';
FLUSH PRIVILEGES;


-- select user :
CREATE USER 'dairy_viewer'@'%' IDENTIFIED BY 'Viewer@123';
GRANT SELECT
ON milk_nest.*
TO 'dairy_viewer'@'%';
FLUSH PRIVILEGES;
